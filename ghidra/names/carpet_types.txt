# Struct and global types for carpet.exe (applied by ApplyTypes.java). Evidence: docs/ENGINE.md.
# Offsets hex. Field types: u8 u16 u32 i8 i16 i32 char, struct names, '*' pointer, '[n]' array.

struct TableRec 0xe                     # class/model dispatch record (14 bytes)
  0x0 parent u32                        # always 0x943cc
  0x4 index u16                         # state (Table A) or model (Table B)
  0x6 handler u32                       # code pointer
  0xa enabled u32

struct ClassEntry 0x12                  # 18-byte class table record, base 0x943da + class*0x12
  0x0 table_a TableRec*                 # update handlers by state (+0x46)
  0x4 table_b TableRec*                 # constructors by model (+0x41)
  0x8 parent u32
  0xc ordinal u16
  0xe zero u32

struct SpriteDesc 0xe                   # 14-byte sprite descriptor, table at 0x97678 (285 entries)
  0x0 base_sprite u16                   # index into the sprite cache (0x211 sprites)
  0x2 unk2 u16
  0x4 unk4 u16
  0x6 half_xy i16                       # -> thing +0x50/+0x52
  0x8 half_z i16                        # -> thing +0x54 (world size)
  0xa shade_group u8                    # indexes DAT_00093f48 / DAT_00093f4e pixel mode; non-zero = no shadow
  0xb unkb u8
  0xc draw_type u8                      # -> thing +0x59 via DAT_00094344
  0xd unkd u8

struct DamageSlot 0x6                   # pending damage by type (6 slots at thing +0x5a)
  0x0 amount i32
  0x4 attacker u16                      # thing index

struct MoveDesc 0x20                    # movement/AI descriptor shared by creatures and projectiles (thing +0x9c), table at 0x96a10
  0x00 id u16
  0x02 turn_min u16                     # min turn rate (ai_wizard_move_13b10)
  0x04 unk4 u16
  0x06 turn_max u16
  0x08 unk8 u16
  0x0a clear_lo i16                     # ground clearance (creature_check_terrain_102b0 flag 2)
  0x0c clear_hi i16
  0x0e z_step i16
  0x10 unk10 u16
  0x12 unk12 u16
  0x14 terrain_mask u32                 # allowed cell types (terrain_type_mask_at_10480 bit)
  0x18 unk18 u16
  0x1a think_period u16                 # "every N ticks" divisor in the creature handlers
  0x1c sight_radius u16
  0x1e fov u16

struct PosLogEntry 0xe                  # camera log entry: written by player_log_position_3e080, read by render_frame_1fab0 (round 6)

  0xa roll i16                          # P.yaw_rate (banks into the turn)

struct PlayerRec 0x801                  # 8 records at state +0x340b (code addresses fields as state + p*0x801 + 0x340b + off)
  0x000 win_timer u16                   # ticks above the win percentage (won at the 17th, 0x10)
  0x002 status u16                      # 2 won (needs Space), 8 leave the tick loop, 4 lost: with 8 and without 2 = restart in place (round 6: port_game.md)
  0x004 quit u8                         # nonzero ends the whole program loop (1 quit, 2 CD check failed)
  0x005 flags u8                        # bit8 debug overlay (word)
  0x006 active u8
  0x007 index u16                       # own player number
  0x009 is_computer u8                  # 1 = AI wizard
  0x00a thing u16                       # player's Thing index
  0x00e view_entry u16                  # log[] entry the renderer takes the camera from (0x1f)
  0x010 log_count u16                   # entries in log[] (0x20); the position is written to [log_count - 1]
  0x012 tick u32                        # level tick counter of the player
  0x018 cheat u32                       # 0xae89e = 1,000,000 health / mana
  0x01c messages PlayerMsg[8]
  0x23c log_template PosLogEntry        # copied into every log[] entry at level start; zoom (0x248) = 0x80
  0x24a log PosLogEntry[32]             # camera log (round 6: typed; only [log_count - 1] is ever written)
  0x40a name char[0x40]
  0x44a input_mode u8                   # 0 flight 2 spell select 3 text entry
  0x44b unk44b u8                       # cleared for every player every tick
  0x44f p u8[0x3b2]                     # "P" sub-block (Thing+0xa0 points here); P+0x37c = spell_found[24] = GameState+0x3bd6 for player 0
